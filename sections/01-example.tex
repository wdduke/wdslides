\begin{frame}
  \frametitle{元数据驱动}
  \begin{itemize}
    \item 标题、作者、导师、单位、日期和 logo 均集中在 \texttt{wdslides-meta.tex}。
    \item 语言由 \texttt{main.tex} 的 class 选项切换：\texttt{chinese} 或 \texttt{english}。
    \item 原有高亮命令仍可使用：\highlight{主题强调}，\stronghighlight{强强调}。
  \end{itemize}
\end{frame}
